
# Data Module

Public interface for OLTP, OLAP, analytics and ML workloads.
Clients connect to a stable cluster API URL. Interchangeable coordinators accept
requests at that address and route work to workers internally.
All classes below are interface specifications, not implementations.

```python
from typing import Literal

type Row = dict[str, Any]
type Schema = dict[str, type]
type TableFormat = Literal["parquet", "arrow"]
type FileFormat = Literal["parquet", "arrow", "csv", "jsonl"]

class DB:
  analytics: Analytics

  def __init__(self, api_url: str, token: str): ...
  def create_table(self, name: str, schema: Schema, key: str,
                   *, format: TableFormat = "parquet") -> Table: ...
  def read(self, uri: str, *, format: FileFormat, **options: Any) -> Query: ...
  def table(self, name: str) -> Table: ...
  def tables(self) -> list[Table]: ...
  def transaction(self) -> Transaction: ...
  def stream(self, name: str) -> Stream: ...
  def create_stream(self, name: str, partitions: int = 1) -> Stream: ...
  def pipeline(self, name: str) -> Pipeline: ...
  def dataset(self, name: str, source: Query | Table) -> Dataset: ...
  def dashboard(self, name: str) -> Dashboard: ...

class Table:
  name: str
  schema: Schema
  key: str
  format: TableFormat

  def insert(self, row: Row) -> None: ...
  def insert_many(self, rows: Iterable[Row]) -> int: ...
  def get(self, key: Any) -> Row | None: ...
  def update(self, key: Any, values: Row) -> bool: ...
  def upsert(self, row: Row) -> None: ...
  def delete(self, key: Any) -> bool: ...
  def query(self) -> Query: ...
  def where(self, **filters) -> Query: ...
  def index(self, *columns: str, unique: bool = False) -> None: ...
  def changes(self) -> Stream: ...
  def drop(self) -> None: ...

class Query:
  def select(self, *columns: str) -> Query: ...
  def where(self, expr: Expr | None = None, **filters) -> Query: ...
  def join(self, other: Table, on: str, how: str = "inner") -> Query: ...
  def group_by(self, *columns: str) -> Query: ...
  def aggregate(self, **aggregations: Expr) -> Query: ...
  def order_by(self, column: str, desc: bool = False) -> Query: ...
  def limit(self, n: int) -> Query: ...
  def sample(self, n: int) -> Query: ...
  def count(self) -> int: ...
  def collect(self) -> list[Row]: ...
  def batches(self, size: int) -> Iterator[list[Row]]: ...
  def to_arrow(self) -> Any: ...
  def to_pandas(self) -> Any: ...
  def write(self, uri: str, *, format: FileFormat, **options: Any) -> Job: ...
  def explain(self) -> str: ...

class Expr:
  def __and__(self, other: Expr) -> Expr: ...
  def __or__(self, other: Expr) -> Expr: ...
  def __gt__(self, value: Any) -> Expr: ...
  def __lt__(self, value: Any) -> Expr: ...
  def __eq__(self, value: Any) -> Expr: ...

def col(name: str) -> Expr: ...

class Agg:
  def count() -> Expr: ...
  def sum(column: str) -> Expr: ...
  def avg(column: str) -> Expr: ...

class Transaction:
  def __enter__(self) -> Transaction: ...
  def __exit__(self, *args) -> None: ...
  def commit(self) -> None: ...
  def rollback(self) -> None: ...

class Dataset:
  name: str
  version: str

  def snapshot(self, tag: str | None = None) -> Dataset: ...
  def split(self, ratios: tuple[float, ...]) -> tuple[Dataset, ...]: ...
  def shuffle(self, seed: int = 42) -> Dataset: ...
  def shard(self, rank: int, world_size: int) -> Dataset: ...
  def cache(self, tier: str = "auto") -> Dataset: ...
  def prefetch(self, batches: int = 4) -> Dataset: ...
  def batches(self, size: int) -> Iterator[Any]: ...
  def to_torch(self, batch_size: int, workers: int = 0) -> Any: ...

class Analytics:
  def track(self, event: str, user_id: str, properties: Row | None = None) -> None: ...
  def events(self, name: str) -> Query: ...
  def funnel(self, steps: list[str], window: str = "7d") -> Query: ...
  def retention(self, event: str, interval: str = "week") -> Query: ...
  def cohorts(self, conditions: dict) -> Query: ...

class Dashboard:
  name: str
  url: str

  def add(self, title: str, query: Query, chart: str = "line") -> Dashboard: ...
  def publish(self) -> str: ...
```

## Formats and transport

`schema` specifies column types; `format` specifies file encoding. Tables default
to Parquet; `arrow` means Arrow IPC. Record the format in table metadata and block
manifests. CSV and JSONL are import/export formats. `DB.read`, `Query.write` and
`Pipeline.from_storage` share format names and validate format-specific options.

File URIs refer to worker-accessible storage. Authorize their locations; exports
publish a result manifest only after all files succeed. The engine supplies
transaction semantics independently of the file encoding.

The SDK calls the coordinator's authenticated API at one stable URL.
[server.md](server.md) owns transport and routing; [overview.md](overview.md)
shows the workflow. Schemas and expressions use structured wire representations.
The SDK preserves request IDs across safe retries within the advertised retention
window; a lost write response reports an unknown outcome until it is resolved.

## Minimal usage

```python
from meshdb import DB, col

db = DB(api_url="https://db.example.com", token="secret")

users = db.create_table(
  "users", {"id": str, "age": int}, key="id", format="parquet"
)
users.insert({"id": "alice", "age": 24})
users.update("alice", {"age": 25})

results = users.query().where(col("age") > 18).collect()

# Read existing files and export query results through the same cluster API.
incoming = db.read("s3://data/incoming/", format="parquet")
incoming.where(col("age") > 18).write(
  "s3://data/exports/adults/", format="csv", delimiter=","
).result()

db.analytics.track("signup", "alice")
db.analytics.funnel(["signup", "purchase"]).collect()

loader = db.dataset("training", users.query()).cache().to_torch(64)
```

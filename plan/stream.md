
# Stream Module

Durable real-time event streaming, consumer groups, replay, CDC,
batch ingestion and continuous data transformations.

```python
class StreamRecord:
  key: str | None
  data: Row
  timestamp: float
  partition: int
  offset: int

class Stream:
  name: str
  partitions: int
  retention: str

  def publish(self, data: Row, key: str | None = None) -> int: ...
  def publish_many(self, rows: Iterable[Row]) -> None: ...
  def subscribe(self, group: str, from_offset: str = "latest") -> Subscription: ...
  def replay(self, offset: int = 0) -> Iterator[StreamRecord]: ...
  def to_table(self, name: str) -> Table: ...
  def stats(self) -> dict: ...

class Subscription:
  group: str
  offsets: dict[int, int]

  def __iter__(self) -> Iterator[StreamRecord]: ...
  def poll(self, timeout: float | None = None) -> list[StreamRecord]: ...
  def ack(self, record: StreamRecord) -> None: ...
  def commit(self) -> None: ...
  def seek(self, partition: int, offset: int) -> None: ...
  def close(self) -> None: ...

class Pipeline:
  name: str
  status: str

  def from_storage(self, uri: str, format: FileFormat = "parquet",
                     **options: Any) -> Pipeline: ...
  def from_stream(self, stream: Stream) -> Pipeline: ...
  def from_table(self, table: Table) -> Pipeline: ...
  def map(self, fn: Callable[[Row], Row]) -> Pipeline: ...
  def filter(self, fn: Callable[[Row], bool]) -> Pipeline: ...
  def window(self, duration: str) -> Pipeline: ...
  def aggregate(self, **metrics: Expr) -> Pipeline: ...
  def into(self, table: Table, mode: str = "append") -> Pipeline: ...
  def schedule(self, cron: str) -> Pipeline: ...
  def start(self) -> Job: ...
  def stop(self) -> None: ...

class Job:
  id: str
  status: str
  progress: float

  def cancel(self) -> None: ...
  def retry(self) -> None: ...
  def logs(self) -> list[str]: ...
  def result(self) -> Any: ...
```

## Minimal usage

```python
events = db.create_stream("events", partitions=4)

events.publish({"user": "alice", "action": "click"})

for record in events.subscribe("analytics"):
  process(record.data)

db.pipeline("ingest").from_stream(events).into(users).start()

db.pipeline("import").from_storage(
  "s3://bucket/data.parquet", format="parquet"
).into(users).start()

changes = users.changes()
```

`FileFormat` and format-specific options are defined in [data.md](data.md).
The source format is independent of the destination table's storage format.

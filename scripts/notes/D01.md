## Checks

Ran the following exact command from the repository root. It checks every local
Markdown link in the two owned files, including heading fragments; parses shell
snippets without running them; compiles the Python example without executing
or importing it; and checks the file paths and flags used by the commands.
It writes no files and makes no network calls.

```sh
orm/.venv/bin/python - <<'PY'
import ast
from pathlib import Path
import re
import subprocess

files = [Path('README.md'), Path('scripts/notes/D01.md')]
links = shells = examples = 0
for path in files:
    text = path.read_text()
    for target in re.findall(r'\]\(([^)]+)\)', text):
        relative, _, fragment = target.partition('#')
        resolved = path.parent / relative if relative else path
        assert resolved.exists(), (path, target)
        if fragment:
            headings = re.findall(r'^#+ (.+)$', resolved.read_text(), re.M)
            anchors = [re.sub(r'[^\w\- ]', '', h.lower()).replace(' ', '-') for h in headings]
            assert fragment in anchors, (path, target)
        links += 1
    for language, snippet in re.findall(r'```(sh|python)\n(.*?)```', text, re.S):
        if language == 'sh':
            subprocess.run(['sh', '-n'], input=snippet, text=True, check=True)
            shells += 1
        else:
            compile(snippet, str(path) + ':example', 'exec')
            examples += 1
for target in ('orm/tests', 'orm/pyproject.toml', 'server/Cargo.toml',
               'scripts/test_acceptance_support.py', 'scripts/test_server_process.py',
               'scripts/test_acceptance_cancellation.py', 'scripts/test_lifecycle_cleanup.py'):
    assert Path(target).exists(), target
for name in ('http', 'lifecycle', 'concurrency', 'latency'):
    tree = ast.parse(Path('scripts/acceptance_' + name + '.py').read_text())
    flags = {arg.value for node in ast.walk(tree)
             if isinstance(node, ast.Call) and isinstance(node.func, ast.Attribute)
             and node.func.attr == 'add_argument' for arg in node.args
             if isinstance(arg, ast.Constant) and isinstance(arg.value, str)}
    assert {'--binary', '--report'} <= flags, name
    if name in ('http', 'lifecycle'):
        assert '--scenario' in flags, name
print(f'PASS: {links} links, {shells} shell blocks, {examples} Python example; command paths and runner flags')
PY
```
